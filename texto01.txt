
import streamlit as st
import pandas as pd
import plotly.express as px
import numpy as np

st.set_page_config(
    page_title="Titanium | Data Analytics",
    page_icon="📊",
    layout="wide"
)

# ---------- DADOS SINTÉTICOS ----------
@st.cache_data
def gerar_dados():
    np.random.seed(42)
    n = 300

    return pd.DataFrame({
        "id": range(1, n + 1),
        "categoria": np.random.choice(
            ["Tecnologia", "Financeiro", "Educação"],
            n
        ),
        "valor": np.random.uniform(100, 5000, n).round(2),
        "idade": np.random.randint(18, 70, n),
        "status": np.random.choice(
            ["Ativo", "Inativo"], n
        )
    })

st.title("📊 Titanium Data Analytics")
st.caption("Dashboard de análise exploratória de dados sintéticos")

# ---------- IMPORTAÇÃO ----------
arquivo = st.sidebar.file_uploader(
    "Importar arquivo CSV ou Excel",
    type=["csv", "xlsx"]
)

if arquivo:
    try:
        if arquivo.name.endswith(".csv"):
            df = pd.read_csv(arquivo, sep=None, engine="python")
        else:
            df = pd.read_excel(arquivo)
    except Exception as erro:
        st.error(f"Erro ao ler arquivo: {erro}")
        st.stop()
else:
    df = gerar_dados()
    st.sidebar.info("Exibindo dados sintéticos de demonstração.")

# ---------- FILTROS ----------
st.sidebar.header("Filtros")

colunas_categoricas = df.select_dtypes(
    include=["object", "category", "bool"]
).columns.tolist()

if colunas_categoricas:
    coluna_filtro = st.sidebar.selectbox(
        "Filtrar por categoria",
        ["Todas"] + colunas_categoricas
    )

    if coluna_filtro != "Todas":
        opcoes = df[coluna_filtro].dropna().unique().tolist()
        selecionados = st.sidebar.multiselect(
            "Valores permitidos",
            opcoes,
            default=opcoes
        )
        df = df[df[coluna_filtro].isin(selecionados)]

# ---------- INDICADORES ----------
c1, c2, c3, c4 = st.columns(4)

c1.metric("Registros", f"{len(df):,}")
c2.metric("Colunas", len(df.columns))
c3.metric("Células vazias", int(df.isna().sum().sum()))
c4.metric("Duplicados", int(df.duplicated().sum()))

st.divider()

# ---------- GRÁFICOS ----------
numericas = df.select_dtypes(include="number").columns.tolist()

categoricas = df.select_dtypes(
    include=["object", "category", "bool"]
).columns.tolist()

aba1, aba2, aba3 = st.tabs([
    "📈 Gráficos",
    "🔎 Exploração",
    "🧪 Qualidade dos dados"
])

with aba1:
    esquerda, direita = st.columns(2)

    with esquerda:
        if numericas:
            coluna_num = st.selectbox(
                "Variável numérica",
                numericas
            )
            fig = px.histogram(
                df,
                x=coluna_num,
                title=f"Distribuição de {coluna_num}",
                template="plotly_white"
            )
            st.plotly_chart(fig, use_container_width=True)
        else:
            st.info("O arquivo não contém colunas numéricas.")

    with direita:
        if categoricas:
            coluna_cat = st.selectbox(
                "Variável categórica",
                categoricas
            )
            contagem = (
                df[coluna_cat]
                .fillna("Não informado")
                .value_counts()
                .rename_axis(coluna_cat)
                .reset_index(name="Quantidade")
            )
            fig = px.bar(
                contagem,
                x=coluna_cat,
                y="Quantidade",
                title=f"Registros por {coluna_cat}",
                template="plotly_white"
            )
            st.plotly_chart(fig, use_container_width=True)
        else:
            st.info("Não há colunas categóricas.")

with aba2:
    st.subheader("Visualização dos registros")
    st.dataframe(df, use_container_width=True)

    if numericas:
        st.subheader("Estatísticas descritivas")
        st.dataframe(
            df[numericas].describe().T,
            use_container_width=True
        )

with aba3:
    st.subheader("Relatório de qualidade")

    qualidade = pd.DataFrame({
        "Coluna": df.columns,
        "Tipo": [str(t) for t in df.dtypes],
        "Valores ausentes": df.isna().sum().values,
        "% ausentes": (
            df.isna().mean().values * 100
        ).round(2),
        "Valores únicos": [
            df[c].nunique() for c in df.columns
        ]
    })

    st.dataframe(qualidade, use_container_width=True)

    st.subheader("Baixar dados filtrados")
    st.download_button(
        "⬇️ Exportar CSV",
        data=df.to_csv(index=False).encode("utf-8-sig"),
        file_name="dados_analisados.csv",
        mime="text/csv"
    )

st.caption("Titanium Data Analytics | Ambiente demonstrativo")
